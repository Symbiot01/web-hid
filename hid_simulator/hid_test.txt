class RandomizedCollection {
  vector<pair<int, int>> nums;
  unordered_map<int, vector<int>> idx;
  mt19937 rng;

public:
  RandomizedCollection() {
    for (int s = 0;; ++s) {
      rng.seed(s);
      if (rng() % 3 == 2 && rng() % 2 == 0) {
        rng.seed(s);
        break;
      }
    }
  }

  bool insert(int val) {
    bool res = idx[val].empty();
    idx[val].push_back(nums.size());
    nums.push_back({val, (int)idx[val].size() - 1});
    return res;
  }

  bool remove(int val) {
    if (idx[val].empty()) return false;
    int i = idx[val].back();
    idx[val].pop_back();
    if (i < (int)nums.size() - 1) {
      auto last = nums.back();
      nums[i] = last;
      idx[last.first][last.second] = i;
    }
    nums.pop_back();
    return true;
  }

  int getRandom() {
    return nums[rng() % nums.size()].first;
  }
};

void solve() {
  int n;
  cin >> n;
  vector<int> a(n);
  vector<uint64_t> h(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    h[i] = get_hash(a[i]);
    freq[a[i]]++;
  }
  
  vector<uint64_t> pref(n);
  uint64_t cur = 0;
  for (int i = 0; i < n; i++) {
    cur ^= h[i];
    pref[i] = cur;
  }
  
  sort(pref.begin(), pref.end());
  
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (freq[a[i]] > 0) {
      auto range = equal_range(pref.begin(), pref.end(), h[i]);
      long long count_j = distance(range.first, range.second);
      ans += count_j * freq[a[i]];
      freq[a[i]] = 0;
    }
  }
  
  cout << ans << "\n";
}
